<script>
function showreferenceno(errorcode, apiname, expip) {
    var serviceCategory = document.getElementById("ServiceCategory").value;
    var url = '${pageContext.request.contextPath}/Multiple?action=showfailurerefernceno'
        + '&errorcode=' + encodeURIComponent(errorcode)
        + '&apiname=' + encodeURIComponent(apiname)
        + '&expip=' + encodeURIComponent(expip)
        + '&ServiceCategory=' + encodeURIComponent(serviceCategory);
    window.open(url, '_blank');
}
